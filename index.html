                async function pullFromCloud() {

                    if (!cloudUrl.value) return;

                    isSyncing.value = true;

                    try {

                        const res = await fetch(cloudUrl.value, {

                            method: 'GET',

                            headers: { 'Accept': 'application/json, text/plain, */*' }

                        });

                        

                        if (!res.ok) {

                            throw new Error(`HTTP 錯誤狀態碼: ${res.status}`);

                        }


                        const rawText = await res.text();


                        // 判斷是否回傳了 HTML 頁面（代表權限未設定為「所有人」或網址錯誤）

                        if (rawText.trim().startsWith('<') || rawText.includes('<!DOCTYPE html>')) {

                            throw new Error('Google 回傳了登入網頁，請檢查 Google Apps Script 部署權限是否設為「所有人 (Anyone)」');

                        }


                        let data = {};

                        if (rawText.trim()) {

                            try {

                                data = JSON.parse(rawText);

                            } catch (parseErr) {

                                throw new Error('雲端資料格式不正確，無法解析 JSON');

                            }

                        }


                        if (data && typeof data === 'object') {

                            // 若雲端試算表完全是空的，且本地已有資料，則自動備份至雲端

                            if (Object.keys(data).length === 0 && Object.keys(ratingsData.value).length > 0) {

                                await pushToCloud();

                                showToast('已成功初始化雲端資料庫！');

                            } else {

                                ratingsData.value = data;

                                saveStorage(false);

                                showToast('已成功從雲端同步最新分數數據！');

                            }

                        }

                    } catch (e) {

                        console.warn('Cloud pull issue:', e);

                        let errorDetail = e.message || '連線失敗';

                        if (e.name === 'TypeError' && (e.message.includes('fetch') || e.message.includes('Failed'))) {

                            errorDetail = '無法連線至 Google 雲端 API (Failed to fetch)。\n\n可能原因：\n1. 雲端 API 網址填寫錯誤或未包含 /exec\n2. Google Apps Script 部署權限未設為「所有人 (Anyone)」\n3. 正在沙盒/預覽環境中運作（部署至正式網站或 GitHub Pages 後即可正常連線）。';

                        }

                        alert('雲端同步提示：\n' + errorDetail);

                    } finally {

                        isSyncing.value = false;

                    }

                }

                    const headers = ['年級', '班別', ...dateHeaders, '已巡視日數', '跨日累計總分', '日均得分', '級內排名'];

                    const rows = [headers];


                    cumulativeDataList.value.forEach(item => {

                        const dateScoresRow = dateHeaders.map(d => item.dateScores[d] !== undefined ? item.dateScores[d] : '-');

                        rows.push([

                            item.grade,

                            item.name,

                            ...dateScoresRow,

                            item.daysCount,

                            item.totalScore,

                            item.avgPerDay,

                            item.gradeRank

                        ]);

                    });


                    const csvContent = '\uFEFF' + rows.map(e => e.join(',')).join('\n');

                    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });

                    const url = URL.createObjectURL(blob);

                    const link = document.createElement('a');

                    link.setAttribute('href', url);

                    link.setAttribute('download', `晨讀比賽跨日累計總分與每日明細表.csv`);

                    document.body.appendChild(link);

                    link.click();

                    document.body.removeChild(link);

                    showToast('已成功匯出跨日累計總分與每日明細報表！');

                }


                function exportToCSV() {

                    const headers = ['日期', '班別', '年級', '校長評分', '副校長評分', '家長評分', '老師評分', '平均專注度', '評語與備註'];

                    const rows = [headers];


                    allClassList.value.forEach(cls => {

                        const principalScore = getScoreForClassRole(cls.name, '校長') ?? '';

                        const vicePrincipalScore = getScoreForClassRole(cls.name, '副校長') ?? '';

                        const parentScore = getScoreForClassRole(cls.name, '家長') ?? '';

                        const teacherScore = getScoreForClassRole(cls.name, '老師') ?? '';

                        const avgScore = getClassAverageScore(cls.name);


                        const notesArr = roles.map(r => {

                            const tag = getTagForClassRole(cls.name, r);

                            const note = getNoteForClassRole(cls.name, r);

                            if (tag || note) return `${r}:[${tag}]${note}`;

                            return '';

                        }).filter(Boolean);


                        rows.push([

                            selectedDate.value,

                            cls.name,

                            cls.grade,

                            principalScore,

                            vicePrincipalScore,

                            parentScore,

                            teacherScore,

                            avgScore,

                            `"${notesArr.join('; ')}"`

                        ]);

                    });


                    const csvContent = '\uFEFF' + rows.map(e => e.join(',')).join('\n');

                    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });

                    const url = URL.createObjectURL(blob);

                    const link = document.createElement('a');

                    link.setAttribute('href', url);

                    link.setAttribute('download', `晨讀專注評分報表_${selectedDate.value}.csv`);

                    document.body.appendChild(link);

                    link.click();

                    document.body.removeChild(link);

                    showToast('已成功匯出 CSV 報表！');

                }
